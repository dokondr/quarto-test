def ask_llm(model, query:str, text:str = "")->str:
    client = Client(
        host="https://ollama.com",
        headers={'Authorization': 'Bearer ' + os.environ.get('OLLAMA_API_KEY')}
    )
    
    messages = [
      {
        'role': 'user',
        'content': query + '\n' +text, 
      },
    ]

    output = io.StringIO()
    for part in client.chat(model, messages=messages, stream=True):
      # print(part['message']['content'], end='', flush=True)
      print(part['message']['content'], end='', flush=True, file=output)

    result = output.getvalue()
    output.close()
    return result

def create_ontology_instance(model, schema, text):

    prompt = f"""
    You are an information extraction system. 
    Your task is to create ontology instance from the text bellow 
    using provided ontology schema. 
         
    OUTPUT FORMAT (STRICT JSON)
    
    -------------------------
    EXTRACTION RULES
    -------------------------
    
    You MUST extract only what is explicitly or strongly implied in the text. 
    Do not hallucinate entities or relationships.
    
    1. Do not invent entities not grounded in the text.
    2. Preserve original wording when possible for names.
    3. Merge duplicates (same concept = same node).
    4. Abstract ideas clearly but minimally.
    5. Keep descriptions concise and faithful.
    6. If uncertain, prefer omission over speculation.
    7. Relations must include source types, source names and target types, target names.

    Return Json of the following format:
    {{
    "nodes": [ 
        {{
          "type": "Idea", # or Medium, Source, Period, School, etc 
          "name": "...",
          "description": "...",
          ...
        }},
        ...
        {{
          "type": "Medium",  
          "name": "...",
          "description": "...",
          ...
        }}
        ...
        {{
          "type": "School", 
          "name": "...",
          "description": "...",
          ...
        }},
        ...
        {{
          "type": "Period", 
          "name": "...",
          "start_date": "...",
          "end_date": "...",
          ...
        {{
          "type": "Source", 
          "name": "...", # NOT TITLE 
          "description": "...",
          ...

    ]
    "relations": [
        {{
        "type": "..."
        "source_type": <source type>,
        "source_name": <source name>,
        "target_type": <target type>,
        "target_name": <target name>
        }},
        ...
    ]
    }}

    Return description as a single text string, do not put it in any list!
    Do not concatenate words with underscore (_) in relation types!
    In 'Source' entity (node) use attribute 'name' insttead of 'title'! 
    Return clean Json. Do not prefix your result with others text, 
    such as "```json" or something else!   
    
    Ontology schema: 
    {schema}
    
    Text:
    {text}
    
    """

    return ask_llm(model, prompt)
